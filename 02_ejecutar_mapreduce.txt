# ------------------------------------------------------------------------------
# 2.1. Cargar el archivo limpio en HDFS y comprobar su presencia
# ------------------------------------------------------------------------------
hdfs dfs -mkdir -p /data/netflix/clean/
hdfs dfs -put -f netflix_clean.csv /data/netflix/clean/
hdfs dfs -ls -h /data/netflix/clean/

# ------------------------------------------------------------------------------
# 2.3. Ejecutar el trabajo MapReduce mediante Hadoop Streaming
#      Entrada: /data/netflix/clean/netflix_clean.csv
#      Salida:  /data/netflix/results/actores/
# ------------------------------------------------------------------------------
# Eliminar carpeta de salida previa si existiera (Hadoop requiere que no exista)
hdfs dfs -rm -r -f /data/netflix/results/actores/
hdfs dfs -mkdir -p /data/netflix/results/

# Dar permisos de ejecución al mapper y al reducer
chmod +x 02_mapper_actores.py 02_reducer_actores.py

# Comando completo de ejecución MapReduce con Hadoop Streaming (Hadoop 3.x)
hadoop jar $HADOOP_HOME/share/hadoop/tools/lib/hadoop-streaming-*.jar \
  -files 02_mapper_actores.py,02_reducer_actores.py \
  -mapper "python3 02_mapper_actores.py" \
  -reducer "python3 02_reducer_actores.py" \
  -input /data/netflix/clean/netflix_clean.csv \
  -output /data/netflix/results/actores/

# ------------------------------------------------------------------------------
# Verificar archivos generados en HDFS y descargar partes para exportar CSV
# ------------------------------------------------------------------------------
hdfs dfs -ls -h /data/netflix/results/actores/
mkdir -p actores
hdfs dfs -get -f /data/netflix/results/actores/part-* actores/
python3 02_exportar_resultado.py
